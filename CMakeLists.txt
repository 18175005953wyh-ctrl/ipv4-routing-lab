cmake_minimum_required(VERSION 3.20)
project(ipv4-routing-lab VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(routing_core src/ipv4.c src/route.c src/route_reader.c)
target_include_directories(routing_core PUBLIC include)
add_executable(ipv4-routing-lab src/main.c)
target_link_libraries(ipv4-routing-lab PRIVATE routing_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/test_runner.c)
    target_link_libraries(core_tests PRIVATE routing_core)
    add_test(NAME core_checks COMMAND core_tests)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cli_checks COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py"
        "$<TARGET_FILE:ipv4-routing-lab>" "${CMAKE_CURRENT_SOURCE_DIR}/examples/routes.csv")
    set_tests_properties(core_checks cli_checks PROPERTIES TIMEOUT 90)
endif()
